extends SceneTree
## Story 021 frame time: canteen with a full wave spawned among the tables; averages physics/process time.
var f: int = 0
var game: Node
var phys: Array[float] = []
var proc: Array[float] = []
func _initialize() -> void:
	var kinds: Array[EnemyData] = []
	var pos: PackedVector3Array = PackedVector3Array()
	var m: EnemyData = load("res://assets/data/enemies/melee.tres")
	var pi: EnemyData = load("res://assets/data/enemies/ranged_pistol.tres")
	var sg: EnemyData = load("res://assets/data/enemies/ranged_shotgun.tres")
	var b: EnemyData = load("res://assets/data/enemies/boss.tres")
	var xs: Array[float] = [-3.4, 0.2, 3.6]
	var i: int = 0
	for z: float in [-9.0, -1.0, 8.5, 0.0]:
		for x: float in xs:
			kinds.append(m if i < 8 else (pi if i < 10 else sg))
			pos.append(Vector3(x, 0.05, z))
			i += 1
	kinds.append(b); pos.append(Vector3(0.2, 0.05, -12.0))
	game = (load("res://src/levels/test_canteen_push_table.tscn") as PackedScene).instantiate()
	game.set(&"debug_spawn_kinds", kinds)
	game.set(&"debug_spawn_positions", pos)
	game.set(&"debug_override_player_pose", true)
	game.set(&"debug_player_position", Vector3(0.2, 0.05, 15.0))
	game.set(&"debug_keep_heart_rate", 70.0)
	root.add_child(game)
func _process(_d: float) -> bool:
	f += 1
	if f > 60 and f <= 660:
		phys.append(Performance.get_monitor(Performance.TIME_PHYSICS_PROCESS) * 1000.0)
		proc.append(Performance.get_monitor(Performance.TIME_PROCESS) * 1000.0)
	if f == 660:
		var moving: int = 0
		var moved: int = 0
		for p: PushableProp in PushableProp.get_all(self):
			if not p.is_resting(): moving += 1
		var sp: float = 0.0; var sr: float = 0.0; var mp: float = 0.0
		for v: float in phys: sp += v; mp = maxf(mp, v)
		for v: float in proc: sr += v
		var alive: int = get_nodes_in_group(&"enemies").size()
		print("PROBE frametime enemies=%d props_not_resting_at_end=%d physics_ms avg=%.3f max=%.3f process_ms avg=%.3f frames=%d" % [alive, moving, sp / phys.size(), mp, sr / proc.size(), phys.size()])
		quit()
	return false
