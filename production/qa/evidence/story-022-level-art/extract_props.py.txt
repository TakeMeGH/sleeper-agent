import re, json, sys, glob, os
ENVI = "E:/Unity Project/fps-substance/Assets/_Game/Art/envi/"
guid2name = {}
for m in glob.glob(ENVI + "*.fbx.meta"):
    g = re.search(r"guid: ([0-9a-f]+)", open(m).read()).group(1)
    guid2name[g] = os.path.basename(m)[:-9]
def docs(path):
    txt = open(path, encoding="utf-8").read()
    parts = re.split(r"^--- !u!(\d+) &(\d+).*$", txt, flags=re.M)
    out = {}
    for i in range(1, len(parts), 3):
        out[parts[i+1]] = (parts[i], parts[i+2])
    return out
def transforms(d):
    # stripped (non-prefab) Transforms: id -> (pos, rot, father, goname)
    res = {}
    for fid, (cls, body) in d.items():
        if cls == "4" and "m_LocalPosition" in body:
            p = re.search(r"m_LocalPosition: \{x: ([-\d.e]+), y: ([-\d.e]+), z: ([-\d.e]+)\}", body)
            r = re.search(r"m_LocalRotation: \{x: ([-\d.e]+), y: ([-\d.e]+), z: ([-\d.e]+), w: ([-\d.e]+)\}", body)
            f = re.search(r"m_Father: \{fileID: (\d+)\}", body)
            res[fid] = ([float(v) for v in p.groups()], [float(v) for v in r.groups()], f.group(1) if f else "0")
    return res
for scene in sys.argv[1:]:
    d = docs(scene)
    tr = transforms(d)
    out = []
    for fid, (cls, body) in d.items():
        if cls != "1001":
            continue
        g = re.search(r"m_SourcePrefab: \{fileID: \d+, guid: ([0-9a-f]+)", body).group(1)
        name = guid2name.get(g, g)
        vals = {}
        for m in re.finditer(r"propertyPath: (m_Local\w+\.\w)\n\s+value: ([-\d.e]+)", body):
            vals[m.group(1)] = float(m.group(2))
        parent = re.search(r"m_TransformParent: \{fileID: (\d+)\}", body).group(1)
        pp = tr.get(parent)
        pos = [vals.get("m_LocalPosition." + a, 0.0) for a in "xyz"]
        rot = [vals.get("m_LocalRotation." + a, 1.0 if a == "w" else 0.0) for a in "xyzw"]
        scl = [vals.get("m_LocalScale." + a, 1.0) for a in "xyz"]
        out.append(dict(fbx=name, pos=pos, rot_xyzw=rot, scale=scl, parent_pos=pp[0] if pp else None, parent_rot=pp[1] if pp else None))
    out.sort(key=lambda e: (e["fbx"], e["pos"]))
    json.dump(out, open(os.path.join(os.path.dirname(__file__), os.path.basename(scene).replace(".unity", "_props.json")), "w"), indent=1)
    from collections import Counter
    print(scene, Counter(e["fbx"] for e in out))
    pr = set((tuple(e["parent_pos"] or []), tuple(e["parent_rot"] or [])) for e in out)
    print(" parents:", pr)

# ---- world transforms (second pass) ----
def qmul(a, b):
    ax, ay, az, aw = a; bx, by, bz, bw = b
    return [aw*bx + ax*bw + ay*bz - az*by, aw*by - ax*bz + ay*bw + az*bx, aw*bz + ax*by - ay*bx + az*bw, aw*bw - ax*bx - ay*by - az*bz]
def qrot(q, v):
    x, y, z, w = q
    m = [[1-2*(y*y+z*z), 2*(x*y-z*w), 2*(x*z+y*w)], [2*(x*y+z*w), 1-2*(x*x+z*z), 2*(y*z-x*w)], [2*(x*z-y*w), 2*(y*z+x*w), 1-2*(x*x+y*y)]]
    return [sum(m[i][j]*v[j] for j in range(3)) for i in range(3)], m
for scene in sys.argv[1:]:
    d = docs(scene); tr = transforms(d)
    def world_of(fid, pos, rot):
        while fid in tr:
            ppos, prot, father = tr[fid]
            v, _ = qrot(prot, pos)
            pos = [ppos[i] + v[i] for i in range(3)]; rot = qmul(prot, rot); fid = father
        return pos, rot
    out = []
    for fid, (cls, body) in d.items():
        if cls != "1001":
            continue
        g = re.search(r"m_SourcePrefab: \{fileID: \d+, guid: ([0-9a-f]+)", body).group(1)
        vals = {m.group(1): float(m.group(2)) for m in re.finditer(r"propertyPath: (m_Local\w+\.\w)\n\s+value: ([-\d.e]+)", body)}
        parent = re.search(r"m_TransformParent: \{fileID: (\d+)\}", body).group(1)
        pos = [vals.get("m_LocalPosition." + a, 0.0) for a in "xyz"]
        rot = [vals.get("m_LocalRotation." + a, 1.0 if a == "w" else 0.0) for a in "xyzw"]
        wp, wr = world_of(parent, pos, rot)
        _, m = qrot(wr, [0, 0, 0])
        basis = ", ".join("%.4f" % m[r][c] for c in range(3) for r in range(3))  # Godot Transform3D column-major x,y,z axes
        out.append(dict(fbx=guid2name.get(g, g), world_pos=[round(v, 4) for v in wp], world_rot_xyzw=[round(v, 5) for v in wr],
                        godot_transform="Transform3D(%s, %.4f, %.4f, %.4f)" % (basis, *wp)))
    out.sort(key=lambda e: (e["fbx"], e["world_pos"]))
    json.dump(out, open(os.path.join(os.path.dirname(__file__), os.path.basename(scene).replace(".unity", "_world.json")), "w"), indent=1)
