extends SceneTree
func _initialize() -> void:
	var arena: Node3D = (load("res://src/waves/test_arena.tscn") as PackedScene).instantiate()
	arena.set("start_delay", 100000.0)
	arena.set("log_events", false)
	root.add_child(arena)
	_run.call_deferred(arena)
func _run(arena: Node3D) -> void:
	for i in 5: await physics_frame
	var sp: WaveSpawner = arena.get_node("WaveSpawner")
	sp.stop()
	var player: Node3D = get_first_node_in_group("player")
	var heart: HeartRate = player.get_node("HeartRate")
	var m: EnemyBase = sp.spawn_enemy(sp.data.get_enemy_data(&"melee"), Vector3(0, 0, 8), 1)
	var states: Array = []
	var rates: Array = []
	var last: StringName = &""
	for f in 360:
		await physics_frame
		player.global_position = Vector3.ZERO
		if m.get_state() != last:
			last = m.get_state()
			states.append("%s@%d d=%.2f" % [last, f, Vector2(m.global_position.x, m.global_position.z).length()])
		rates.append(heart.get_rate())
	var jumps: Array = []
	for i in range(1, rates.size()):
		if absf(rates[i] - rates[i - 1]) > 1.0:
			jumps.append("f%d %.1f->%.1f" % [i, rates[i - 1], rates[i]])
	print("PROBE melee states=%s" % [states])
	print("PROBE melee player heart jumps (take_hit -> apply_hit, damage %.1f)=%s" % [m.get_damage(), jumps])
	quit(0)
